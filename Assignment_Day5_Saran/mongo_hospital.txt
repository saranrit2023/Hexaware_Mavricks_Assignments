use hospital_clinical_db

db.createCollection("clinical_notes", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["visitId", "patientId", "doctorName", "diagnosis", "recordedAt"],
      properties: {
        visitId: { bsonType: "int" },
        patientId: { bsonType: "int" },
        doctorName: { bsonType: "string" },
        diagnosis: { bsonType: "string" },
        recordedAt: { bsonType: "date" }
      }
    }
  }
});
