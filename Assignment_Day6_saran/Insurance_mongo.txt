db.claims_notes.aggregate([
  {
    $group: {
      _id: "$status",
      totalClaims: { $sum: 1 }
    }
  },
  { $sort: { totalClaims: -1 } }
]);


db.claims_notes.aggregate([
  { $match: { status: "Settled", settledDate: { $ne: null } } },
  {
    $project: {
      claimId: 1,
      damageCategory: 1,
      settlementDays: {
        $divide: [
          { $subtract: ["$settledDate", "$filedDate"] },
          1000 * 60 * 60 * 24 // Convert milliseconds to days
        ]
      }
    }
  },
  {
    $group: {
      _id: null,
      settledCount: { $sum: 1 },
      avgSettlementDays: { $avg: "$settlementDays" }
    }
  }
]);


db.claims_notes.aggregate([
  {
    $group: {
      _id: "$damageCategory",
      incidentCount: { $sum: 1 }
    }
  },
  { $sort: { incidentCount: -1 } },
  { $limit: 3 }
]);
