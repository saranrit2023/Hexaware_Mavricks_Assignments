db.usage_events.aggregate([
  {
    $group: {
      _id: { $hour: "$timestamp" },
      eventCount: { $sum: 1 },
      totalDataMB: { $sum: "$dataMB" },
      totalDurationSecs: { $sum: "$sessionSeconds" }
    }
  },
  { $sort: { _id: 1 } }
]);

db.usage_events.aggregate([
  {
    $group: {
      _id: "$eventType",
      sessionCount: { $sum: 1 },
      avgSessionSeconds: { $avg: "$sessionSeconds" }
    }
  },
  {
    $project: {
      _id: 0,
      eventType: "$_id",
      sessionCount: 1,
      avgSessionSeconds: { $round: ["$avgSessionSeconds", 2] }
    }
  }
]);


db.usage_events.aggregate([
  {
    $group: {
      _id: "$towerId",
      totalEvents: { $sum: 1 },
      totalDataMB: { $sum: "$dataMB" }
    }
  },
  { $sort: { totalEvents: -1, totalDataMB: -1 } },
  { $limit: 1 }
]);
